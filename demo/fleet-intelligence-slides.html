Shared repository

One parameterized query; PostgreSQL chooses the access path

WITH query_vector AS (
  SELECT azure_openai.create_embeddings(
    %s, %s
  )::vector(1536) AS embedding
),
candidates AS MATERIALIZED (
  SELECT
    s.*,
    s.embedding <=> query_vector.embedding
      AS vector_distance,
    ST_Distance(...) / 1000 AS distance_km
  FROM horizon_ship.shipments AS s
  CROSS JOIN query_vector
  WHERE (%s IS NULL OR s.status = %s)
    AND (%s IS NULL OR s.eta BETWEEN ...)
    AND (%s IS NULL OR ST_DWithin(...))
  ORDER BY s.embedding <=> query_vector.embedding
  LIMIT 100
)
SELECT *,
  0.72 * (1 - vector_distance)
  + 0.28 * spatial_score AS hybrid_score
FROM candidates
ORDER BY hybrid_score DESC;
